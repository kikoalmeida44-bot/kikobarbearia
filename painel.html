const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbwoFpOt-eXHQ-9utcqPp3R3i6eow2PYxvsSZKuJKiYEziND_xU6vz3kgA2q5zY4MePc/exec'; async function atualizarPainelAdmin() { try { const response = await fetch(SCRIPT_URL + '?acao=dashboard'); const data = await response.json(); // 1. Atualiza os Ganhos Estimados const elGanhos = document.getElementById('ganhos-estimados') || document.querySelector('.ganhos-valor'); if (elGanhos) { elGanhos.innerText = data.faturamento.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); } // 2. Atualiza a Quantidade de Agendamentos const elAgendamentos = document.getElementById('total-agendamentos') || document.querySelector('.agendamentos-qtd'); if (elAgendamentos) { elAgendamentos.innerText = data.agendamentosMes; } // 3. Atualiza a Lista de Clientes no Top Fidelidade const elFidelidade = document.getElementById('lista-fidelidade') || document.querySelector('.fidelidade-container'); if (elFidelidade) { if (data.fidelidade && data.fidelidade.length > 0) { elFidelidade.innerHTML = data.fidelidade.map(c => `
🏆 ${c.nome} - ${c.visitas} agendamento(s)
` ).join(''); } else { elFidelidade.innerHTML = `

Nenhum cliente registado este mês.

`; } } } catch (err) { console.error('Erro ao carregar dados do painel:', err); } } // Executa a atualização do painel assim que a página é aberta document.addEventListener('DOMContentLoaded', atualizarPainelAdmin);
